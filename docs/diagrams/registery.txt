@startuml

package "definition" {
  ' --- Interfaces ---
  interface NetabaseDefinition {
    type TreeNames: NetabaseDefinitionTreeNames<Self>;
    type DefKeys: NetabaseDefinitionKeys<Self>;
    type DebugName: Clone + Debug + PartialEq + Eq + Hash + Serialize;
    type SubscriptionKeysDiscriminant: Debug;
    type SubscriptionKeys: NetabaseDefinitionSubscriptionKeys<Discriminant = Self::SubscriptionKeysDiscriminant>;
    const SUBSCRIPTION_REGISTRY: DefinitionSubscriptionRegistry<'static, Self>;

    debug_name() -> Self::DebugName;
    definition_name() -> &'static str;
    schema() -> DefinitionSchema;
    export_toml() -> String;
  }

  interface NetabaseDefinitionTreeNames<D: NetabaseDefinition> {
    get_tree_names(D::Discriminant) -> Vec<Self>;
    get_model_tree_names(&self) -> Option<&'static ModelTreeNames<'static, D, M>;
  }

  interface NetabaseDefinitionKeys<D: NetabaseDefinition>
}

package "subscription" {
  interface NetabaseDefinitionSubscriptionKeys

  ' --- Registry Types ---
  struct "DefinitionSubscriptionRegistry<'a, D: NetabaseDefinition>" as DefinitionSubscriptionRegistry {
    +entries: &'a [SubscriptionEntry<&'static str, D::Discriminant>];
    +const new(&'a [SubscriptionEntry<&'static str, D::Discriminant>]) -> Self;
    +get_subscribers(&self, topic: &str) -> Option <&[D::Discriminant]>;
    +model_subscribes_to(&self, topic: &str, model: D::Discriminant) -> bool;
    +get_model_topics(&self, model: D::Discriminant);
  }

  struct SubscriptionEntry<TopicDiscriminant, ModelDiscriminant: 'static> {
    +topic: TopicDiscriminant,
    +subscribers: &'static [ModelDiscriminant]
  }
}

package "schema" {
  ' --- Schema & Export Types ---

  struct RepositorySchema {
    +schema_format_version: u32;
    +name: String;
    +definitions: Vec<DefinitionSchema>;
    +to_toml() -> String;
    +from_toml(str) -> Result<Self, String>;
    +compute_hash() -> u64;
  }

  struct DefinitionSchema {
    +schema_format_version: u32;
    +name: String;
    +models: Vec<ModelSchema>;
    +structs: Vec<StructSchema>;
    +subscriptions: Vec<String>;
    +model_history: Vec<ModelVersionHistory>;
    +schema_hash: Option<u64>;
    +config: Option<DefinitionConfig>;
    +compare(&self, other) -> SchemaComparisonResult;
  }

  struct DefinitionConfig {
    +retention_days: Option<u32>;
    +compression_enabled: bool;
    +max_blob_size: Option<u64>;
    +auto_migration: bool;
    +metadata: HashMap<String, String>;
  }

  enum SchemaComparisonResult {
      Identical,
      LocalNewer { families: Vec<(String, u32, u32)> },
      RemoteNewer { families: Vec<(String, u32, u32)> },
      Mixed { local: Vec<...>, remote: Vec<...> },
      Conflict { families: Vec<(String, u32)> },
  }

  ' --- Model Types ---

  struct ModelSchema {
    +name: String;
    +fields: Vec<FieldSchema>;
    +subscriptions: Vec<String>;
    +subscription_immutable: bool;
    +family: Option<String>;
    +version: Option<u32>;
    +is_current: bool;
    +is_libp2p_enabled: bool;
    +is_content_addressed: bool;
    +content_addressed_config: Option<ContentAddressedConfig>;
  }

  struct ContentAddressedConfig {
    +hasher: String;
    +function: String;
    +key_type: Option<String>;
  }

  struct StructSchema {
    +name: String;
    +fields: Vec<StructFieldSchema>;
    +is_tuple: bool;
  }

  struct StructFieldSchema {
    +name: String;
    +type_name: String;
  }

  struct FieldSchema {
    +name: String;
    +type_name: String;
    +key_type: KeyTypeSchema;
  }

  enum KeyTypeSchema {
      Primary,
      Secondary,
      Relational { definition: String, model: String },
      Blob,
      Regular,
  }

  ' --- Versioning & Migration Types ---

  struct ModelVersionHistory {
    +family: String;
    +current_version: u32;
    +versions: Vec<VersionedModelSchema>;
    +migration_paths: Vec<MigrationPathSchema>;
  }

  struct VersionedModelSchema {
    +version: u32;
    +struct_name: String;
    +fields: Vec<FieldSchema>;
    +subscriptions: Vec<String>;
    +subscription_immutable: bool;
    +version_hash: u64;
    +supports_downgrade: bool;
    +supports_upgrade: bool;
    +is_libp2p_enabled: bool;
    +is_content_addressed: bool;
    +content_addressed_config: Option<ContentAddressedConfig>;
  }

  struct MigrationPathSchema {
    +from_version: u32;
    +to_version: u32;
    +may_lose_data: bool;
    +field_changes: Vec<FieldChangeSchema>;
  }

  enum FieldChangeSchema {
      Added { name, type_name, has_default },
      Removed { name, type_name },
      Renamed { old_name, new_name },
      TypeChanged { name, old_type, new_type },
  }
}

package "redb_definition" {
  interface RedbDefinition {
    type ModelTableDefinition<'db>;
    type ReadOnlyTables;
    type RecordIter<'a>;
    detect_versions(db: &Database) -> Result<Vec<DetectedVersion>>;
    migrate_all(db: &Database, options: &MigrationOptions) -> Result<MigrationResult>;
    init_tables(db: &Database) -> Result<()>;
    ...()
  }

  struct MigrationResult {
    +records_migrated: usize
    +records_failed: usize
    +errors: Vec<String>
    +migrations_performed: Vec<(String, u32, u32)>
  }

  struct MigrationOptions {
    +continue_on_error: bool
    +max_errors: usize
    +dry_run: bool
    +delete_old_tables: bool
  }

  struct DetectedVersion {
    +family: String
    +version: u32
    +table_name: String
    +record_count: u64
  }
}
